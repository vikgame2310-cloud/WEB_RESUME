class ConnectManager {
    var connect_from_cr_con = mutableListOf()

    fun add(connect: Connect) {
        connect_from_cr_con.add(connect)
    }

    fun getId(): Int {
        if (connect_from_cr_con.isEmpty()) {
            return 1
        }

        return connect_from_cr_con.maxOf { it.id } + 1
    }

        println("id\t|\tsender\t|\trecipient\t|\themet\t|\tsizeKb\t|\tdata-time")

            println("${connect.id}\t|\t${connect.sender}\t|\t${connect.recipient}\t|\t${connect.theme}\t|\t${connect.sizeKb}\t|\t${connect.createdAt}")
        }
    }

        if (x ==1){
            if (y == 1){
                connect_from_cr_con.sortBy { it.createdAt }
            }
            else if(y == 2){
                connect_from_cr_con.sortByDescending { it.createdAt }
            }
        }
        else if (x == 2){
            if (y == 1){
                connect_from_cr_con.sortBy { it.sender }
            }
            else if(y == 2){
                connect_from_cr_con.sortByDescending { it.sender }
            }
        }
        else if (x == 3){
            if (y == 1){
                connect_from_cr_con.sortBy { it.recipient }
            }
            else if(y == 2){
                connect_from_cr_con.sortByDescending { it.recipient }
            }
        }
        else if (x == 4){
            if (y == 1){
                connect_from_cr_con.sortBy { it.theme }

            }
            else if(y == 2){
                connect_from_cr_con.sortByDescending { it.theme }
            }
        }

    }

    fun remove_list(del: Int, x: List = connect_from_cr_con): Boolean{
        for (connect in connect_from_cr_con) {
            if (connect.id == del) {
                connect_from_cr_con.remove(connect)

            }
        }
        return false
    }

            if (connect.id == change_Id) {

fun createParticipant(): String{
    var sender: String
    val tipe_platform = listOf("gmail.com", "yandex.ru")
    while (true) {
        sender = readLine().toString()
        if (sender.substringAfter("@") in tipe_platform &&
            sender.substringBefore("@").isNotEmpty()) {
            return sender
        }
        else{println("Не верно указан пользователь, повторите снова")}
    }
}


fun createSizeKb(): Int {
    while (true) {
        print("размер в КБ: ")

        val sizeKb = readlnOrNull()?.toIntOrNull()

        if (sizeKb != null && sizeKb >= 0) {
            return sizeKb
        }

        println("Размер письма должен быть неотрицательным числом. Повторите ввод.")
    }
}


fun saveToJson() {
    val json = buildString {
        append("[\n")

        for ((index, connect) in manager.connect_from_cr_con.withIndex()) {
            append("  {\n")
            append("    \"id\": ${connect.id},\n")
            append("    \"sender\": \"${connect.sender}\",\n")
            append("    \"recipient\": \"${connect.recipient}\",\n")
            append("    \"theme\": \"${connect.theme}\",\n")
            append("    \"sizeKb\": ${connect.sizeKb},\n")
            append("    \"data-time\": \"${connect.createdAt}\"\n")
            append("  }")

            if (index != manager.connect_from_cr_con.lastIndex) {
                append(",")
            }

            append("\n")
        }

        append("]")
    }

    java.io.File("connections.json").writeText(json)
    println("Данные сохранены в JSON-файл")
}


fun readJson(){
    val file = java.io.File("connections.json")

    if (!file.exists()) {
        println("JSON-файл не найден")
        return
    }

    manager.connect_from_cr_con.clear()

    var lines = file.readLines()

    var id = 0
    var sender = ""
    var recipient = ""
    var theme = ""
    var sizeKb = 0
    var createdAt = ""

    for (line in lines) {
        if (line.trim().startsWith("\"id\"")) {
            id = line.substringAfter(":")
                .replace("\"", "")
                .replace(",", "")
                .trim()
                .toInt()
        }
        if (line.trim().startsWith("\"sender\"")) {
            sender = line.substringAfter(":")
                .replace("\"", "")
                .replace(",", "")
                .trim()
        }
        if (line.trim().startsWith("\"recipient\"")){
            recipient = line.substringAfter(":")
                .replace("\"", "")
                .replace(",", "")
                .trim()
        }
        if (line.trim().startsWith("\"theme\"")){
            theme = line.substringAfter(":")
                .replace("\"", "")
                .replace(",", "")
                .trim()
        }
        if (line.trim().startsWith("\"sizeKb\"")) {
            sizeKb = line.substringAfter("\"sizeKb\":")
                .replace("\"", "")
                .replace(",", "")
                .trim()
                .toInt()
        }
        if (line.trim().startsWith("\"data-time\"")){
            createdAt = line.substringAfter("\"data-time\":")
                .replace("\"", "")
                .replace(",", "")
                .trim()
            manager.add(Connect(id, sender, recipient, theme, sizeKb, createdAt))
        }

    }
    println("файл загружен")
}


fun saveToCsv() {
    val csv = buildString {
        append("id,sender,recipient,theme,sizeKb,data-time\n")

        for (connect in manager.connect_from_cr_con) {
            append("${connect.id},${connect.sender},${connect.recipient},${connect.theme}," +
                    "${connect.sizeKb},${connect.createdAt}\n")
        }
    }
    java.io.File("connections.csv").writeText(csv)
    println("Данные сохранены в CSV-файл")
}


fun readCsv() {
    val file = java.io.File("connections.csv")

    if (!file.exists()) {
        println("CSV-файл не найден")
        return
    }

    manager.connect_from_cr_con.clear()

    var lines = file.readLines()

    var id = 0
    var sender = ""
    var recipient = ""
    var theme = ""
    var sizeKb = 0
    var createdAt = ""

    for (line in 1 until lines.size) {
        var level = lines[line].split(",")
        id = level[0].toInt()
        sender = level[1]
        recipient = level[2]
        theme = level[3]
        sizeKb = level[4].toInt()
        createdAt = level[5]
        manager.add(Connect(id, sender, recipient, theme, sizeKb, createdAt))
    }
    println("CSV-файл загружен")
}


fun statistics() {
    if (manager.connect_from_cr_con.isEmpty()) {
        println("Список почтовых отправлений пуст")
        return
    }

    val sizes = manager.connect_from_cr_con.map { it.sizeKb }

    println("Агрегированные показатели по размеру писем:")
    println("Количество писем: ${manager.connect_from_cr_con.size}")
    println("Суммарный размер писем: ${sizes.sum()} КБ")
    println("Средний размер письма: ${sizes.average()} КБ")
    println("Минимальный размер письма: ${sizes.minOrNull()} КБ")
    println("Максимальный размер письма: ${sizes.maxOrNull()} КБ")
}

fun filter_pro(chose: String, chose_1: String, rezult_filtr: MutableList){
    for (connect in manager.connect_from_cr_con) {
        if (connect in rezult_filtr) { continue }
        if (
            chose == "id" && connect.id.toString() == chose_1 ||
            chose == "sender" && connect.sender == chose_1 ||
            chose == "recipient" && connect.recipient == chose_1 ||
            chose == "theme" && connect.theme == chose_1 ||
            chose == "sizeKb" && connect.sizeKb.toString() == chose_1 ||
            chose == "data-time" && connect.createdAt == chose_1
        ) {
            rezult_filtr.add(connect)
        }
    }

    var go_to: Int?
    do {
        println("выберите действия:")
        println("1. вывести результат")
        println("2. присвоить главному списку")
        println("3. продолжить фильтрацию")
